# -------------------- MATERIALS ------------------------------

def make_mat(name, color, metallic=0.0, roughness=0.65):
    mat = bpy.data.materials.new(name)
    mat.diffuse_color = (*color, 1)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get("Principled BSDF")
    bsdf.inputs["Base Color"].default_value = (*color, 1)
    bsdf.inputs["Metallic"].default_value = metallic
    bsdf.inputs["Roughness"].default_value = roughness
    return mat

cream = make_mat("Warm ivory walls", (0.78, 0.72, 0.60))
white = make_mat("Ceiling white", (0.88, 0.88, 0.84))

# -------------------- PRIMITIVES -----------------------------

def cube_obj(name, loc, scale, mat, bevel=0.0):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc)
    obj = bpy.context.object
    obj.name = name
    obj.dimensions = scale
    bpy.ops.object.transform_apply(
        location=False, rotation=False, scale=True
    )
    if mat:
        obj.data.materials.append(mat)
    if bevel > 0:
        mod = obj.modifiers.new("Soft edges", "BEVEL")
        mod.width = bevel
        mod.segments = 2
    return obj

def cylinder_obj(name, loc, radius, depth, mat,
                 vertices=16, rotation=None):
    bpy.ops.mesh.primitive_cylinder_add(
        vertices=vertices,
        radius=radius,
        depth=depth,
        location=loc
    )
    obj = bpy.context.object
    obj.name = name
    if rotation:
        obj.rotation_euler = rotation
    if mat:
        obj.data.materials.append(mat)
    return obj

def text_obj(body, loc, size, mat, rotation=(0, 0, 0)):
    curve = bpy.data.curves.new("Text_" + body, "FONT")
    curve.body = body
    curve.size = size
    curve.extrude = 0.003
    obj = bpy.data.objects.new(body, curve)
    bpy.context.collection.objects.link(obj)
    obj.location = loc
    obj.rotation_euler = rotation
    obj.data.materials.append(mat)
    return obj

# -------------------- SEATING: 204 CHAIRS --------------------

# 12 rows x 17 seats = 204 seats
# Eight chairs on left, nine on right of the centre aisle.

ROWS = 12
SEATS_PER_ROW = 17
SEAT_SPACING = 0.67
ROW_SPACING = 0.78
FIRST_ROW_Y = -4.35

def create_chair(x, y, row, number):
    # Individual seat
    cube_obj(f"Seat_R{row:02d}_N{number:02d}",
             (x, y, 0.78),
             (0.51, 0.48, 0.14), red, 0.055)

    # Backrest
    cube_obj(f"Back_R{row:02d}_N{number:02d}",
             (x, y + 0.19, 1.12),
             (0.51, 0.13, 0.68), red, 0.055)

    # Four supports
    for dx in (-0.19, 0.19):
        for dy in (-0.16, 0.16):
            cube_obj("Chair metal support",
                     (x + dx, y + dy, 0.42),
                     (0.045, 0.045, 0.65), black)

    # Armrests
    for side in (-1, 1):
        cube_obj("Chair armrest",
                 (x + side*0.28, y, 0.98),
                 (0.065, 0.42, 0.08), black, 0.025)

for row in range(ROWS):
    y = FIRST_ROW_Y + row * ROW_SPACING

    # Left block: 8 chairs
    for seat in range(8):
        x = -6.2 + seat * SEAT_SPACING
        create_chair(x, y, row+1, seat+1)

    # Right block: 9 chairs
    for seat in range(9):
        x = 1.75 + seat * SEAT_SPACING
        create_chair(x, y, row+1, seat+9)

# -------------------- CEILING LIGHTS ------------------------

def add_area_light(name, loc, power, size, color):
    data = bpy.data.lights.new(name, type='AREA')
    data.energy = power
    data.shape = 'RECTANGLE'
    data.size = size
    data.size_y = 1.0
    data.color = color
    obj = bpy.data.objects.new(name, data)
    bpy.context.collection.objects.link(obj)
    obj.location = loc
    obj.rotation_euler = (0, 0, 0)
    return obj

def aim_at(obj, target):
    direction = Vector(target) - obj.location
    obj.rotation_euler = direction.to_track_quat('-Z', 'Y').to_euler()
