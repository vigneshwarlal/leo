# Materials
def material(name, color, roughness=0.7, metallic=0):
    m = bpy.data.materials.new(name)
    m.diffuse_color = (*color, 1)
    m.use_nodes = True
    p = m.node_tree.nodes.get("Principled BSDF")
    p.inputs["Base Color"].default_value = (*color, 1)
    p.inputs["Roughness"].default_value = roughness
    p.inputs["Metallic"].default_value = metallic
    return m

white = material("White concrete", (0.9, 0.89, 0.82))

# Basic objects
def cube(name, loc, size, mat, bevel=0):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc)
    o = bpy.context.object
    o.name = name
    o.dimensions = size
    bpy.ops.object.transform_apply(
        location=False, rotation=False, scale=True
    )
    if mat:
        o.data.materials.append(mat)
    if bevel:
        b = o.modifiers.new("Edge bevel", 'BEVEL')
        b.width = bevel
        b.segments = 2
        o.modifiers.new("Weighted normals", 'WEIGHTED_NORMAL')
    return o

def sphere(name, loc, scale, mat):
    bpy.ops.mesh.primitive_uv_sphere_add(
        segments=12, ring_count=8, location=loc
    )
    o = bpy.context.object
    o.name = name
    o.scale = scale
    o.data.materials.append(mat)
    return o

def beam(name, a, b, radius, mat):
    a, b = Vector(a), Vector(b)
    d = b - a
    bpy.ops.mesh.primitive_cylinder_add(
        vertices=8, radius=radius, depth=d.length,
        location=(a+b)/2
    )
    o = bpy.context.object
    o.name = name
    o.rotation_mode = 'QUATERNION'
    o.rotation_quaternion = d.to_track_quat('Z', 'Y')
    o.data.materials.append(mat)
    return o

def text_object(body, loc, size, mat):
    c = bpy.data.curves.new("Sign text", 'FONT')
    c.body = body
    c.size = size
    c.align_x = 'CENTER'
    o = bpy.data.objects.new(body, c)
    bpy.context.collection.objects.link(o)
    o.location = loc
    o.rotation_euler[0] = math.pi/2
    c.extrude = 0.002
    c.materials.append(mat)
    return o

def shrub(x, y, size=1):
    sphere("Bush", (x, y, 0.35*size),
           (0.55*size, 0.45*size, 0.4*size), green)
    for i in range(4):
        a = i*math.pi/2
        sphere("Bush leaf cluster",
               (x+math.cos(a)*0.25*size,
                y+math.sin(a)*0.22*size, 0.4*size),
               (0.33*size, 0.30*size, 0.3*size),
               green2 if i%2 else green)

positions = [
    (-10,-5), (-8,-4), (-6,-5), (-9,0), (-6,0),
    (-4,-1), (4,-1), (6,0), (9,0), (6,-5),
    (8,-4), (10,-5), (-11,-10), (-7,-11),
    (-4,-11), (4,-11), (7,-11), (11,-10),
    (-14,1), (14,1)
]

for x, y in positions:
    shrub(x+random.uniform(-0.3,0.3),
          y+random.uniform(-0.3,0.3),
          random.uniform(0.7,1.2))

def tree(x, y, h=7, spread=2):
    trunk_h = h*0.55

    bpy.ops.mesh.primitive_cylinder_add(
        vertices=10, radius=0.18, depth=trunk_h,
        location=(x, y, trunk_h/2)
    )
    bpy.context.object.name = "Tree trunk"
    bpy.context.object.data.materials.append(bark)

    for i in range(4):
        z = trunk_h + i*h*0.09
        r = spread*(1-i*0.16)
        sphere("Tree canopy", (x, y, z),
               (r, r*0.85, h*0.18),
               green2 if i%2 else green)

tree(-15, -2, 8, 2.6)
