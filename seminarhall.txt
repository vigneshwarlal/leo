# 2. MATERIALS
def material(name, color, roughness=0.7, metallic=0):
    m = bpy.data.materials.new(name)
    m.diffuse_color = (*color, 1)
    m.use_nodes = True
    bs = m.node_tree.nodes.get("Principled BSDF")
    bs.inputs["Base Color"].default_value = (*color, 1)
    bs.inputs["Roughness"].default_value = roughness
    bs.inputs["Metallic"].default_value = metallic
    return m

# 3. GEOMETRY HELPERS
def cube(name, loc, size, mat, bevel=0):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc)
    o = bpy.context.object
    o.name = name
    o.dimensions = size
    bpy.ops.object.transform_apply(
        location=False, rotation=False, scale=True
    )
    if mat:
        o.data.materials.append(mat)
    if bevel:
        mod = o.modifiers.new("Rounded edges", "BEVEL")
        mod.width = bevel
        mod.segments = 2
        o.modifiers.new("Weighted normals", "WEIGHTED_NORMAL")
    return o

def cylinder(name, loc, radius, depth, mat, vertices=16):
    bpy.ops.mesh.primitive_cylinder_add(
        vertices=vertices, radius=radius, depth=depth,
        location=loc
    )
    o = bpy.context.object
    o.name = name
    if mat:
        o.data.materials.append(mat)
    return o

def beam(name, a, b, radius, mat):
    a, b = Vector(a), Vector(b)
    d = b - a
    o = cylinder(name, (a+b)/2, radius, d.length, mat, 10)
    o.rotation_mode = 'QUATERNION'
    o.rotation_quaternion = d.to_track_quat('Z', 'Y')
    return o

def sphere(name, loc, scale, mat):
    bpy.ops.mesh.primitive_uv_sphere_add(
        segments=12, ring_count=8, location=loc
    )
    o = bpy.context.object
    o.name = name
    o.scale = scale
    o.data.materials.append(mat)
    bpy.ops.object.shade_smooth()
    return o

# 5. WINDOWS AND RED/WHITE CURTAINS
def window(side, y):
    x = side * 5.91
    z = 2.15
    ww, hh = 1.45, 1.45

    cube("Window glass", (x, y, z), (.045, ww, hh), glass)

    for yy in [y-ww/2, y+ww/2]:
        cube("Window frame", (x+side*.035, yy, z),
             (.07, .05, hh+.08), white)

    for zz in [z-hh/2, z+hh/2]:
        cube("Window frame", (x+side*.035, y, zz),
             (.07, ww+.08, .05), white)

    for offset, mat in [
        (-ww*.62, red), (-ww*.46, cream),
        (ww*.46, cream), (ww*.62, red)
    ]:
        cube("Curtain", (x-side*.09, y+offset, z),
             (.08, ww*.16, hh+.20), mat, .02)

    beam("Curtain rod",
         (x-side*.09, y-ww*.82, 3.10),
         (x-side*.09, y+ww*.82, 3.10),
         .025, metal)

for s in [-1, 1]:
    for y in [-3.6, -1.0, 1.6, 3.8]:
        window(s, y)

# 9. CLASSROOM CHAIRS WITH WRITING TABLETS
def chair(x, y, number):
    # Seat
    cube(f"Chair {number} seat", (x, y, .49),
         (.51, .46, .10), black, .04)

    # Backrest
    cube(f"Chair {number} back", (x, y+.19, .91),
         (.51, .10, .68), black, .05)

    # Metal legs
    for dx in [-.19, .19]:
        for dy in [-.16, .16]:
            beam("Chair metal leg",
                 (x+dx, y+dy, .10),
                 (x+dx*.9, y+dy*.85, .45),
                 .023, metal)

    # Back support
    for dx in [-.18, .18]:
        beam("Back support", (x+dx, y+.17, .45),
             (x+dx, y+.19, 1.16), .019, metal)

    # Wooden writing tablet
    tx, ty = x+.34, y-.02
    cube(f"Chair {number} writing tablet",
         (tx, ty, .82), (.47, .38, .06), wood, .025)

    beam("Tablet support", (x+.21, y+.02, .47),
         (tx+.1, ty, .79), .022, metal)

# Six rows of nine chairs
rows, cols = 6, 9

    y = -2.55 + r*1.10

        if x > 0:
            x += .22
        x += random.uniform(-.025, .025)
        chair(x, y+random.uniform(-.02, .02), r*cols+c+1)

# 11. CEILING FANS
def fan(x, y):
    z = 3.32
    cylinder("Fan motor", (x, y, z), .14, .20, metal)
    cylinder("Fan hub", (x, y, z-.13), .10, .08, metal)

    for i in range(3):
        angle = math.tau*i/3
        b = cube("Fan blade",
                 (x+math.cos(angle)*.55,
                  y+math.sin(angle)*.55, z-.15),
                 (1.0, .15, .035), metal, .025)
        b.rotation_euler[2] = angle

for x, y in [(-3, -.4), (0, .8), (3, 1.6)]:
    fan(x, y)

# 16. LIGHTING
def area_light(name, loc, power, size):
    bpy.ops.object.light_add(type='AREA', location=loc)
    l = bpy.context.object
    l.name = name
    l.data.energy = power
    l.data.shape = 'DISK'
    l.data.size = size
    l.rotation_euler = (0, 0, 0)
    direction = Vector((0, 0, 1.0))-l.location
    l.rotation_euler = direction.to_track_quat('-Z', 'Y').to_euler()
