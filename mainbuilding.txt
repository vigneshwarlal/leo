def material(name, color, roughness=0.8):
    m = bpy.data.materials.new(name)
    m.diffuse_color = (*color, 1)
    m.use_nodes = True
    m.node_tree.nodes["Principled BSDF"].inputs[
        "Base Color"].default_value = (*color, 1)
    m.node_tree.nodes["Principled BSDF"].inputs[
        "Roughness"].default_value = roughness
    return m

def cube(name, loc, size, mat, bevel=0):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc)
    o = bpy.context.object
    o.name = name
    o.dimensions = size
    bpy.ops.object.transform_apply(
        location=False, rotation=False, scale=True)
    o.data.materials.append(mat)
    if bevel:
        mod = o.modifiers.new("Soft edges", "BEVEL")
        mod.width = bevel
        mod.segments = 2
        o.modifiers.new("Corner normals", "WEIGHTED_NORMAL")
    return o

def cylinder(name, loc, radius, depth, mat, vertices=16):
    bpy.ops.mesh.primitive_cylinder_add(
        vertices=vertices, radius=radius,
        depth=depth, location=loc)
    o = bpy.context.object
    o.name = name
    o.data.materials.append(mat)
    return o

def sphere(name, loc, scale, mat):
    bpy.ops.mesh.primitive_uv_sphere_add(
        segments=16, ring_count=12, location=loc)
    o = bpy.context.object
    o.name = name
    o.scale = scale
    o.data.materials.append(mat)
    bpy.ops.object.shade_smooth()
    return o

def beam(name, a, b, radius, mat):
    a, b = Vector(a), Vector(b)
    direction = b - a
    o = cylinder(name, (a+b)/2, radius, direction.length, mat, 8)
    o.rotation_mode = 'QUATERNION'
    o.rotation_quaternion = direction.to_track_quat('Z', 'Y')
    return o

# Dome mesh
def dome(name, cx, cy, base, radius, height, mat):
    verts, faces = [], []
    n, rings = 48, 24
    for j in range(rings+1):
        t = math.pi/2 * j/rings
        r = radius * math.cos(t)
        z = base + height * math.sin(t)
        for i in range(n):
            a = 2*math.pi*i/n
            verts.append((cx+r*math.cos(a),
                          cy+r*math.sin(a), z))
    for j in range(rings):
        for i in range(n):
            a = j*n+i
            b = j*n+(i+1)%n
            c = (j+1)*n+(i+1)%n
            d = (j+1)*n+i
            faces.append((a,b,c,d))
    mesh = bpy.data.meshes.new(name)
    mesh.from_pydata(verts, [], faces)
    mesh.materials.append(mat)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    for p in mesh.polygons:
        p.use_smooth = True
    solid = obj.modifiers.new("Dome shell", "SOLIDIFY")
    solid.thickness = 0.08
    return obj

def shrub(x,y,size):
    for i in range(5):
        ox = random.uniform(-0.25,0.25)*size
        oy = random.uniform(-0.18,0.18)*size
        r = random.uniform(0.20,0.36)*size
        sphere("Garden shrub", (x+ox,y+oy,0.2+r*0.3),
               (r*1.3,r,r*0.8), leaves)

for side in [-1,1]:
    for i in range(8):
        shrub(side*(4.0+i*0.8),-3.25,0.8)

def tree(x,y,height,radius):
    cylinder("Tree trunk", (x,y,height*0.38),
             0.14,height*0.76,trunkmat,10)
    for i in range(9):
        ox = random.uniform(-0.5,0.5)*radius
        oy = random.uniform(-0.5,0.5)*radius
        oz = random.uniform(-0.4,0.5)*radius
        sphere("Tree foliage",
               (x+ox,y+oy,height*0.75+oz),
               (radius*0.65,radius*0.60,radius*0.75),
               leaves)
